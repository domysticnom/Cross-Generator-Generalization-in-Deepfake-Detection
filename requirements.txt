# ---------------------------------------------------------------------------
# PyTorch is installed SEPARATELY from a CUDA wheel index, NOT from plain PyPI.
# Pick the wheel that matches the card you are on:
#
#   school VM (V100 / older drivers) -- this is what the committed runs used:
#     pip install torch==2.4.1 torchvision==0.19.1 \
#         --index-url https://download.pytorch.org/whl/cu121
#
#   RTX 50-series (Blackwell, sm_120):
#     pip install torch==2.9.* torchvision==0.24.* torchaudio==2.9.* \
#         --index-url https://download.pytorch.org/whl/cu128
#
# The default PyPI wheel ships without sm_120 kernels and fails on Blackwell.
# The pipeline itself is not version-locked: train.py / evaluate.py run on
# either stack, and checkpoints written under one load under the other.
# Run check_env.py after installing; it verifies CUDA actually executes.
# ---------------------------------------------------------------------------

# Core numerics / data handling
# numpy capped below 2: mediapipe + tensorflow on the school VM are compiled
# against numpy 1.x and crash under numpy 2 (numpy.dtype size changed).
numpy>=1.26,<2

opencv-python>=4.10,<5       # opencv 5 requires numpy>=2; use opencv-python-headless on headless machines
protobuf>=3.20.3,<5          # tensorflow 2.16 (via mediapipe) rejects protobuf 5+; keep in the 4.x window
